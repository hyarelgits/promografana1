    - name: installed httpd

    - name: start httpd service

loop.yml

--
- name: understading variables
  hosts: all
  become: true
  vars:
    installed_list:
      - apache2
      - git
      - tree
      - docker
  tasks:
    - name: installed multiple packages at a time
      apt:
        name: "{{ item }}"
        state: present
      loop: "{{ installed_list }}"
